<script>

  const continueBtn = document.getElementById("continueBtn");

  const introContainer = document.getElementById("introContainer");

  const mainContainer = document.getElementById("mainContainer");


  continueBtn.addEventListener("click", () => {

    introContainer.style.display = "none";

    mainContainer.style.display = "flex";

  });


  const noBtn = document.getElementById("noBtn");

  const yesBtn = document.getElementById("yesBtn");

  const hint = document.getElementById("hint");

  const emoji = document.getElementById("emoji");

  const card = document.getElementById("card");

  const reveal = document.getElementById("reveal");

  const typedText = document.getElementById("typedText");

  const music = document.getElementById("bgMusic");


  let attempts = 0;


  const hints = [

    "Hmm… interesting choice 😏",

    "Kannu ma’am, wrong answer 🙄",

    "That button is decorative 😌",

    "Be serious now 💖",

    "Okay you’re just teasing 😈"

  ];


  const emojis = ["💗", "😌", "🙄", "🥺", "😈"];


  function dodge() {

    attempts++;

    noBtn.style.left = Math.random() * 65 + "%";

    noBtn.style.top = Math.random() * 65 + "%";

    noBtn.style.transform = `scale(${Math.max(0.45, 1 - attempts * 0.1)})`;

    hint.textContent = hints[Math.min(attempts, hints.length - 1)];

    emoji.textContent = emojis[Math.min(attempts, emojis.length - 1)];

  }


  noBtn.addEventListener("mouseover", dodge);

  noBtn.addEventListener("touchstart", dodge);


  const lines = [

    "Block mat kar dena bss 😂😂"

  ];


  let line = 0;

  let char = 0;


  function typeLine() {

    if (line < lines.length) {

      if (char < lines[line].length) {

        typedText.innerHTML += lines[line].charAt(char);

        char++;

        setTimeout(typeLine, 40);

      } else {

        line++;

        char = 0;

      }

    }

  }


  const canvas = document.getElementById("confetti");

  const ctx = canvas.getContext("2d");

  canvas.width = window.innerWidth;

  canvas.height = window.innerHeight;


  let confetti = [];


  function startConfetti() {

    for (let i = 0; i < 120; i++) {

      confetti.push({

        x: Math.random() * canvas.width,

        y: Math.random() * canvas.height,

        r: Math.random() * 6 + 4,

        d: Math.random() * 30,

        color: `hsl(${Math.random() * 360},100%,70%)`

      });

    }

    drawConfetti();

  }


  function drawConfetti() {

    ctx.clearRect(0, 0, canvas.width, canvas.height);

    confetti.forEach(p => {

      ctx.beginPath();

      ctx.fillStyle = p.color;

      ctx.arc(p.x, p.y, p.r, 0, Math.PI * 2);

      ctx.fill();

      p.y += Math.cos(p.d) + 2;

      if (p.y > canvas.height) p.y = -10;

    });

    requestAnimationFrame(drawConfetti);

  }


  yesBtn.addEventListener("click", () => {

    card.style.display = "none";

    reveal.style.display = "block";

    setTimeout(() => {

      music.play();

      typeLine();

      startConfetti();

    }, 300);

  });

</script>
